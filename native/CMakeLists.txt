cmake_minimum_required(VERSION 3.24)
project(ScadaNative VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
option(SCADA_USE_VCPKG "Use installed dependencies (recommended on Windows)" ${WIN32})
option(SCADA_SANITIZE "Enable address/undefined sanitizers" OFF)
if(SCADA_USE_VCPKG)
  find_path(ASIO_INCLUDE_DIR asio.hpp REQUIRED)
  find_package(nlohmann_json CONFIG REQUIRED)
  find_package(unofficial-sqlite3 CONFIG REQUIRED)
  set(SCADA_SQLITE unofficial::sqlite3::sqlite3)
else()
  FetchContent_Declare(asio URL https://codeload.github.com/chriskohlhoff/asio/tar.gz/refs/tags/asio-1-30-2 URL_HASH SHA256=755bd7f85a4b269c67ae0ea254907c078d408cce8e1a352ad2ed664d233780e8 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.11.3 URL_HASH SHA256=0d8ef5af7f9794e3263480193c491549b2ba6cc74bb018906202ada498a79406 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(asio json)
  set(ASIO_INCLUDE_DIR ${asio_SOURCE_DIR}/asio/include)
  # Build the amalgamation from the official SQLite source, not a third-party mirror.
  FetchContent_Declare(sqlite_source URL https://codeload.github.com/sqlite/sqlite/tar.gz/refs/tags/version-3.50.4 URL_HASH SHA256=74ed9d2e5930d79564d92b838a61f943a9df01403b51f479fb64ce9aa5dca70d DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(sqlite_source)
  if(NOT EXISTS ${sqlite_source_BINARY_DIR}/sqlite3.c)
    file(MAKE_DIRECTORY ${sqlite_source_BINARY_DIR})
    find_program(MAKE_EXECUTABLE make REQUIRED)
    execute_process(COMMAND ${sqlite_source_SOURCE_DIR}/configure --disable-shared --disable-readline WORKING_DIRECTORY ${sqlite_source_BINARY_DIR} COMMAND_ERROR_IS_FATAL ANY OUTPUT_QUIET)
    execute_process(COMMAND ${MAKE_EXECUTABLE} -j2 sqlite3.c WORKING_DIRECTORY ${sqlite_source_BINARY_DIR} COMMAND_ERROR_IS_FATAL ANY OUTPUT_QUIET)
  endif()
  add_library(scada_sqlite STATIC ${sqlite_source_BINARY_DIR}/sqlite3.c)
  target_include_directories(scada_sqlite PUBLIC ${sqlite_source_BINARY_DIR})
  target_compile_definitions(scada_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0)
  set_target_properties(scada_sqlite PROPERTIES POSITION_INDEPENDENT_CODE ON)
  set(SCADA_SQLITE scada_sqlite)
endif()
find_package(Threads REQUIRED)
add_library(scada_core STATIC src/model.cpp src/protocol.cpp src/archive.cpp src/engine.cpp src/secs.cpp)
target_include_directories(scada_core PUBLIC include)
target_include_directories(scada_core SYSTEM PUBLIC ${ASIO_INCLUDE_DIR})
target_compile_definitions(scada_core PUBLIC ASIO_STANDALONE ASIO_NO_DEPRECATED)
target_link_libraries(scada_core PUBLIC nlohmann_json::nlohmann_json ${SCADA_SQLITE} Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(scada_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(WIN32)
  target_compile_definitions(scada_core PUBLIC _WIN32_WINNT=0x0A00 NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(scada_core PUBLIC ws2_32 mswsock winmm)
endif()
if(MSVC)
  target_compile_options(scada_core PRIVATE /W4 /permissive- /EHsc /utf-8)
else()
  target_compile_options(scada_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(SCADA_SANITIZE AND NOT MSVC)
  target_compile_options(scada_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(scada_core PUBLIC -fsanitize=address,undefined)
endif()
add_library(scada_engine SHARED src/c_api.cpp)
target_compile_definitions(scada_engine PRIVATE SCADA_EXPORTS)
target_link_libraries(scada_engine PRIVATE scada_core)
add_executable(scada_gateway src/main.cpp)
target_link_libraries(scada_gateway PRIVATE scada_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(scada_tests tests/tests.cpp)
  target_link_libraries(scada_tests PRIVATE scada_core)
  add_test(NAME scada_unit COMMAND scada_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME scada_protocol_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:scada_gateway>)
    add_test(NAME scada_ui_bridge COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/c_api.py $<TARGET_FILE:scada_engine>)
    set_tests_properties(scada_protocol_integration scada_ui_bridge PROPERTIES TIMEOUT 60)
  endif()
endif()
